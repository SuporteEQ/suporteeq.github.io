# Configure os botoes do modo manual neste arquivo.
# Cada bloco [botao] cria um botao na janela.
# Linhas vazias e linhas que comecam com # sao ignoradas.
# Comentarios no fim da linha nao sao removidos.
#
# Campos principais:
# titulo=Texto do botao
# comando=Comando a executar. Pode repetir ou usar comando={ ... }.
# Varios comandos no mesmo botao rodam no mesmo terminal, em sequencia.
# Arquivos .bat/.cmd recebem call automaticamente; .ps1 roda via powershell.exe.
#
# Campos opcionais:
# verificar=Caminho que deve existir antes de executar o botao.
# Nao use verificar= para arquivo que sera baixado/criado pelo proprio comando.
# admin=true para executar como outro usuario via PsExec, sem elevacao garantida.
# Usa C:\temp\user_a.txt e C:\temp\user_b.txt como credenciais do PsExec.
# elevate=true para executar com UAC nativo, digitando/escolhendo a credencial no prompt.
# Se admin=true e elevate=true estiverem juntos, elevate=true prevalece e usa UAC.
# Se user_a.txt tiver apenas Administrador, usa %COMPUTERNAME%\Administrador.
# Logs ficam em C:\temp\logs; .cmd temporarios ficam em C:\temp\cmd.
# Em admin=true/elevate=true, o .cmd temporario e apagado em caso de sucesso.
# Em admin=false, .cmd temporarios sao apagados apos sucesso e preservados em falhas.
# admin=false para executar sem elevacao. Este e o padrao.
# mensagem=Texto exibido apos iniciar os comandos.
# Em admin=true/elevate=true, a mensagem aparece depois que a execucao terminar.
# Comandos admin=true/elevate=true devem ser nao interativos; saida/erro ficam em log temporario.
#
# Variaveis disponiveis:
# %SCRIPT_DIR% = pasta onde estao estes arquivos
# %AUTO_BAT% = caminho completo do auto.bat
#
# Exemplo completo comentado:
# [botao]
# titulo=Exemplo completo
# admin=true
# elevate=true
# verificar=%AUTO_BAT%
# verificar=C:\temp\tests\test.bat
# comando=echo Este comando roda no mesmo terminal do proximo comando.
# comando="%AUTO_BAT%"
# comando=C:\temp\tests\test.bat argumento1 argumento2
# comando=C:\temp\tests\test.ps1 -Nome "Aluno 1" -Force
# comando=pause
# mensagem=Todos os comandos do exemplo terminaram.
#
# Exemplo usando elevacao UAC:
# [botao]
# titulo=Teste elevado
# elevate=true
# comando=C:\temp\tests\test.ps1
#
# Exemplo usando bloco de comandos:
# [botao]
# titulo=Exemplo com bloco
# admin=false
# comando={
# echo Primeiro comando
# echo Segundo comando
# pause
# }

[botao]
titulo=Registrar data/hora
admin=false
comando="%AUTO_BAT%"
verificar=%AUTO_BAT%
mensagem=Data/hora registrada em C:\temp\auto.txt.


[botao]
titulo=mostrar auto.txt
admin=false
comando=type C:\Temp\auto.txt
comando=pause

[botao]
titulo=Abrir notepad
admin=false
comando=notepad.exe

[botao]
titulo=Abrir Sublime
admin=false
comando=subl.exe

[botao]
titulo=(local) test.bat
admin=false
comando=C:\temp\tests\test.bat
verificar=C:\temp\tests\test.bat

[botao]
titulo=(local) test.ps1
admin=false
comando=C:\temp\tests\test.ps1
verificar=C:\temp\tests\test.ps1

[botao]
titulo=(local/admin) test.ps1
admin=true
comando=C:\temp\tests\test.ps1
verificar=C:\temp\tests\test.ps1

[botao]
titulo=(local/admin/elevate) test.ps1
admin=true
elevate=true
comando=C:\temp\tests\test.ps1
verificar=C:\temp\tests\test.ps1
